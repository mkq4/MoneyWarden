например виджет dashboard

widget/dashboard/ui/index.ts // barrel импорт

// Если можно выделить компоненты котоыре будут использоваться в Dashboard

widget/dashboard/components/index.ts // barrel импорты компонентов
widget/dashboard/components/SummaryCards.tsx // компонент
widget/dashboard/components/RecentTransactions.tsx // компонент
... // сколь угодно компонентов

widget/dashboard/index.ts // barrel import
